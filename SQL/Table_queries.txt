-- Source documents (tickets, resolutions, KB, SOP)
CREATE TABLE Documents (
    Id BIGINT IDENTITY(1,1) CONSTRAINT PK_Documents PRIMARY KEY,
    DocType NVARCHAR(50) NOT NULL,      -- 'ticket','resolution','kb','sop','log'
    Title NVARCHAR(400) NULL,
    Content NVARCHAR(MAX) NOT NULL,
    SourceUri NVARCHAR(500) NULL,
    Module NVARCHAR(100) NULL,          -- 'tagging','validation','rendering', etc.
    Tags NVARCHAR(400) NULL,            -- pipe or comma separated tags
    AclLabel NVARCHAR(100) NULL DEFAULT 'support',
    CreatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    UpdatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);

-- Chunked text (RAG units)
CREATE TABLE DocChunks (
    Id BIGINT IDENTITY(1,1) CONSTRAINT PK_DocChunks PRIMARY KEY,
    DocId BIGINT NOT NULL FOREIGN KEY REFERENCES Documents(Id) ON DELETE CASCADE,
    ChunkIndex INT NOT NULL,
    ChunkText NVARCHAR(MAX) NOT NULL,
    Metadata NVARCHAR(MAX) NULL,
    -- Store vector as VARBINARY. We'll compute similarity in C# for simplicity.
    Embedding VARBINARY(MAX) NULL,
    CreatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);

-- Stores token -> index mapping and IDF weights
IF OBJECT_ID('TfidfVocab') IS NULL
BEGIN
  CREATE TABLE TfidfVocab (
    Term NVARCHAR(100) NOT NULL PRIMARY KEY,
    TermIndex INT NOT NULL UNIQUE,
    Df INT NOT NULL,
    Idf FLOAT NOT NULL
  );
END

-- Stores a single row with vector size & versioning (optional but helpful)
IF OBJECT_ID('TfidfMeta') IS NULL
BEGIN
  CREATE TABLE TfidfMeta (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    VocabSize INT NOT NULL,
    BuiltAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    Notes NVARCHAR(200) NULL
  );
END
